<!-- JsonStore 契约 -->
class JsonStore:
    def __init__(self, path: str)            # path = data.json 绝对路径
    def load(self) -> dict                    # 返回 {"profiles": [...], "rooms": [...]}；缺失/损坏时回退空 schema
    def save(self, data: dict) -> None        # 原子写：data.json.tmp → os.replace

<!-- API 路由契约（ReminderHandler 分发） -->
GET    /api/profiles         -> 200 {"profiles": [Profile]}
POST   /api/profiles         -> body {"name","template"} -> 201 {"profile": Profile} | 400 {"error"}
PUT    /api/profiles/<id>    -> body {"name","template"} -> 200 {"profile": Profile} | 404
DELETE /api/profiles/<id>    -> 204 | 404
GET    /api/rooms            -> 200 {"rooms": []}   <!-- Plan 02 填满 -->
POST/PUT/DELETE /api/rooms/* -> Plan 02 实现

<!-- 数据模型 -->
Profile = {id: "p_"+uuid4hex12, name: str, template: str, created_at: ISO8601, updated_at: ISO8601}
Room    = {id: "r_"+uuid4hex12, number: str, category: "class"|"vip"}

<!-- 占位符常量 -->
PLACEHOLDERS = ["{时间}", "{教室号}", "{老师}"]

<!-- 静态路由（白名单，防路径穿越） -->
STATIC_ROUTES = {"/": "index.html", "/index.html": "index.html", "/style.css": "style.css", "/app.js": "app.js"}

<!-- app.js 关键函数（Plan 02 复用 renderRooms / openRoomEditor） -->
api(path, options), renderProfiles(), renderRooms(), openProfileEditor(mode, profile?),
openRoomEditor(mode, room?), escapeHtml(s), templateToHtml(template), serializeComposer(editor)

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-01 | Injection (XSS) | template 渲染进列表/预览 | mitigate | templateToHtml 先 escapeHtml 全部文本，仅把三个白名单 token {时间}/{教室号}/{老师} 替换为 chip span；严禁把未转义的用户文本塞进 innerHTML |

| T-01-05 | Elevation of privilege | 无鉴权 | accept | 单用户本地工具，localhost-only；本地用户本可读写自己的 data.json，无提权面 |
| T-01-SC | Tampering | npm/pip/cargo installs | accept | 本项目零第三方依赖（纯标准库），无供应链面 |
